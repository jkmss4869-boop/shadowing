function doGet(e) {
  var action = e.parameter.action;
  var sheetName = e.parameter.sheetName || "Sheet1";
  
  // 1. API Lấy từ vựng: ?action=getVocab&sheetName=Unit1
  if (action === "getVocab") {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = sheet.getDataRange().getValues();
    var headers = data[0]; // Giả sử dòng 1 là header: word, ipa, level, type, meaning, context
    var vocabList = [];
    
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] !== "") { // Nếu cột word không rỗng
        vocabList.push({
          word: String(data[i][0]),
          ipa: String(data[i][1] || ""),
          level: String(data[i][2] || ""),
          type: String(data[i][3] || ""),
          meaning: String(data[i][4] || ""),
          context: String(data[i][5] || "")
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify(vocabList))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (action === "getVideoVocab") {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var displayData = sheet.getDataRange().getDisplayValues();
    var rawData = sheet.getDataRange().getValues();
    var vocabList = [];
    
    // Quét từ dòng 2 (bỏ qua Header dòng 1) - Map đúng 6 cột tối giản
    for (var i = 1; i < displayData.length; i++) {
      if (displayData[i][0] !== "") { // Nếu Start Time không rỗng
        vocabList.push({
          startTime: String(displayData[i][0] || rawData[i][0] || ""),
          stopTime: String(displayData[i][1] || rawData[i][1] || ""),
          context: String(displayData[i][2] || ""),
          word: String(displayData[i][3] || ""),
          meaning: String(displayData[i][4] || ""),
          videoId: String(displayData[i][5] || "").trim(), // Lấy ID ở cột F
          bgUrl: String(displayData[i][6] || rawData[i][6] || "").trim() // Lấy URL Ảnh Nền ở cột G
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify(vocabList))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // API câu hỏi Nghe hiểu. Dòng 1 là tiêu đề, các cột lần lượt là:
  // A: câu hỏi | B: đáp án đúng | C-F: 4 phương án trả lời.
  if (action === "getListeningComprehension") {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var data = sheet.getDataRange().getDisplayValues();
    var questions = [];

    for (var i = 1; i < data.length; i++) {
      var question = String(data[i][0] || "").trim();
      var correctAnswer = String(data[i][1] || "").trim();
      var options = [data[i][2], data[i][3], data[i][4], data[i][5]]
        .map(function (option) { return String(option || "").trim(); })
        .filter(function (option) { return option !== ""; });

      // Bỏ qua dòng trống hoặc dòng chưa nhập đủ dữ liệu câu hỏi.
      if (question && correctAnswer && options.length >= 2) {
        questions.push({
          question: question,
          correctAnswer: correctAnswer,
          options: options
        });
      }
    }

    return ContentService.createTextOutput(JSON.stringify(questions))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var action = data.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 2. API Lưu điểm tổng
    if (action === "saveResults") {
      var scoreSheet = ss.getSheetByName("Scores") || ss.insertSheet("Scores");
      if (scoreSheet.getLastRow() === 0) {
        scoreSheet.appendRow(["Timestamp", "Student name", "Lesson", "Game Type", "Score", "Total"]);
      }
      scoreSheet.appendRow([
        new Date(), data.name, data.sheetName, data.gameType, data.score, data.total
      ]);

      var detailSheet = ss.getSheetByName("Details") || ss.insertSheet("Details");
      if (detailSheet.getLastRow() === 0) {
        detailSheet.appendRow(["Timestamp", "Student name", "Lesson", "Question", "Target answer", "Student answer", "Correct", "Points", "Wrong speech attempts"]);
      }
      var time = new Date();
      var details = data.details || [];
      for (var i = 0; i < details.length; i++) {
        detailSheet.appendRow([
          time, data.name, data.sheetName, details[i].question, details[i].targetWord,
          details[i].studentAnswer, details[i].isCorrect ? "Correct" : "Wrong",
          details[i].points, details[i].wrongAttempts || "None"
        ]);
      }
      return ContentService.createTextOutput(JSON.stringify({
        status: "success", detailsSaved: details.length
      })).setMimeType(ContentService.MimeType.JSON);
    }
    else if (action === "saveScore") {
      var sheet = ss.getSheetByName("Scores") || ss.insertSheet("Scores");
      // Nếu chưa có header thì tạo header
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["Timestamp", "Tên học sinh", "Bài học", "Game Type", "Điểm đạt", "Tổng điểm"]);
      }
      sheet.appendRow([
        new Date(),
        data.name,
        data.sheetName,
        data.gameType,
        data.score,
        data.total
      ]);
    } 
    // 3. API Lưu chi tiết từng câu
    else if (action === "saveDetailList") {
      var sheet = ss.getSheetByName("Details") || ss.insertSheet("Details");
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["Timestamp", "Tên học sinh", "Bài học", "Ngữ cảnh câu", "Từ chuẩn", "Học sinh trả lời", "Đúng/Sai", "Điểm", "Lịch sử đọc sai (STT)"]);
      }
      
      var time = new Date();
      var details = data.details;
      for (var i = 0; i < details.length; i++) {
        sheet.appendRow([
          time,
          data.name,
          data.sheetName,
          details[i].question,
          details[i].targetWord,
          details[i].studentAnswer,
          details[i].isCorrect ? "Đúng" : "Sai",
          details[i].points,
          details[i].wrongAttempts || "Không có"
        ]);
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
